Query:
  select 'public' as schema, tablename, policyname, roles, cmd, qual, with_check
  from pg_policies where schemaname='public'
  union all
  select 'storage', tablename, policyname, roles, cmd, qual, with_check
  from pg_policies where schemaname='storage' and tablename='objects'
  order by schema, tablename, cmd, policyname;
