posts  -- UPDATED (b): consolidated SELECT, is_visible_profile added,
       -- 'friends' branch dropped. INSERT/UPDATE deliberately left as their
       -- original duplicate pairs (write-side check migration not applied).
  DELETE "You can delete your own posts"  roles={public}
  DELETE "you delete your own moments"  roles={public}
  INSERT "You can post as yourself"  roles={public}
  INSERT "you post your own moments"  roles={authenticated}
  SELECT "posts are readable by their audience"  roles={public}  <-- now the only SELECT policy
    qual: (select auth.uid())=user_id
        OR (visibility='public' AND is_visible_profile(user_id))
        OR (visibility='circle' AND circle_id IS NOT NULL AND (select auth.uid()) IS NOT NULL
            AND is_visible_profile(user_id)
            AND exists(select 1 from circles c where c.id=(posts.circle_id-1000000)
                and (c.visibility = any(array['open_to_read','Open to read'])
                     or c.owner=(select auth.uid())
                     or private.is_circle_member(c.id,(select auth.uid())))))
  UPDATE "You can edit or delete your own posts"  roles={public}
  UPDATE "own posts are editable"  roles={public}
